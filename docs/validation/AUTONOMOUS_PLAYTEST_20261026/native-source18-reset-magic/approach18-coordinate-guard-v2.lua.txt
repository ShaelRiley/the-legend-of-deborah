local function vec(v) return {v.x,v.y,v.z} end
local function entity(e)
    if not IsValid(e) then return nil end
    local a,b=e:GetCollisionBounds()
    return {id=e:EntIndex(),birth=e:GetCreationID(),class=e:GetClass(),position=vec(e:GetPos()),
        worldMins=vec(e:GetPos()+a),worldMaxs=vec(e:GetPos()+b),kind=e.GetBoxKind and e:GetBoxKind() or nil}
end
local function trace(t) return {hit=t.Hit,startSolid=t.StartSolid,allSolid=t.AllSolid,fraction=t.Fraction,position=vec(t.HitPos),entity=entity(t.Entity)} end
local function clear(t) return not t.Hit and not t.StartSolid and not t.AllSolid end

end

    for i=2,#c.points do
        local a,b=c.points[i-1],c.points[i] local raw=sweep(a,b,false)
        local seg={from=vec(a),to=vec(b),fullHull=trace(raw),legacyDefaultFullHull=trace(sweep(a,b,false,true))}

        row.fullSegmentsClear=row.fullSegmentsClear and clear(raw)
        local count=math.max(1,math.ceil(a:Distance(b)/16))
        for j=0,count do
            local v=a+(b-a)*(j/count) local stand=sweep(v,v,false)
            local support=sweep(v+Vector(0,0,4),v-Vector(0,0,6),false)
            local supported=support.Hit and not support.StartSolid and not support.AllSolid and support.HitNormal.z>.7

        end
        row.segments[#row.segments+1]=seg
    end
    row.finiteNativeCenterlineClear=row.fullSegmentsClear and row.standingSamplesClear and row.supportSamplesValid
    r.candidates[#r.candidates+1]=row
end
